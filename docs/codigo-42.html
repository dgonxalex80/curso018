Unidad 4.2 | Intervalos de confianza

Curso 018 - Inferencia Estadística

Código R: intervalos de confianza paramétricos y no paramétricos

Los ejemplos son reproducibles y utilizan principalmente funciones de R base.

1. Cobertura de intervalos para una media

set.seed(18042)
M <- 200; n <- 40; mu <- 30; sigma <- sqrt(10); alpha <- 0.05
mat <- replicate(M, rnorm(n, mean=mu, sd=sigma))
medias <- colMeans(mat)
sd_m <- apply(mat, 2, sd)
error <- qt(1-alpha/2, n-1) * sd_m / sqrt(n)
li <- medias-error; ls <- medias+error
cobertura <- (li <= mu & mu <= ls)
cat("Cobertura empírica:", mean(cobertura), "\n")
Cobertura empírica: 0.95 
plot(seq_len(M), medias, ylim=range(c(li,ls,mu)), pch=16,
     xlab="Muestra", ylab="Intervalo para la media", col="#476885")
segments(seq_len(M),li,seq_len(M),ls,
         col=ifelse(cobertura,"#476885","#C65B4A"))
abline(h=mu,lty=2,lwd=2)

2. Una población: media, proporción y varianza

set.seed(41)
x <- rnorm(35, mean=82, sd=11)
t.test(x, conf.level=0.95)$conf.int

attr(,"conf.level")
[1] 0.95
# Intervalo Wilson para proporción: 47 éxitos de 80 ensayos
prop.test(47,80,correct=FALSE,conf.level=0.95)$conf.int
[1] 0.4780404 0.6889414
attr(,"conf.level")
[1] 0.95
# Intervalo chi-cuadrado para varianza (supone normalidad)
n <- length(x); s2 <- var(x); a <- 0.05
c(inferior=(n-1)*s2/qchisq(1-a/2,n-1),
  superior=(n-1)*s2/qchisq(a/2,n-1))

 98.09833 257.38223

5. Dos proporciones: aproximación normal

x1 <- 65; n1 <- 110; x2 <- 48; n2 <- 105
a <- 0.05; z <- qnorm(1-a/2)
p1 <- x1/n1; p2 <- x2/n2
d <- p1-p2
se <- sqrt(p1*(1-p1)/n1+p2*(1-p2)/n2)
c(inferior=d-z*se,superior=d+z*se)

7. IC no paramétrico exacto/conservador para mediana

set.seed(45)
x <- rexp(31,rate=1/12)
n <- length(x); alpha <- 0.05
# k máximo con cobertura al menos 1-alpha
k_validos <- which(2*pbinom(seq_len(floor(n/2))-1,n,0.5) <= alpha)
if(length(k_validos)==0) stop("Muestra insuficiente para el nivel solicitado")
k <- max(k_validos)
x_ord <- sort(x)
ic_mediana <- c(inferior=x_ord[k], superior=x_ord[n-k+1])
c(ic_mediana, cobertura_teorica=1-2*pbinom(k-1,n,0.5))
         inferior          superior cobertura_teorica 
        3.2009474        11.9797420         0.9705506

8. Bootstrap percentil para una media y una mediana

set.seed(46)
x <- rlnorm(45,meanlog=2,sdlog=0.6)
B <- 4000; alpha <- 0.05
boot_medias <- replicate(B,mean(sample(x,length(x),replace=TRUE)))
boot_medianas <- replicate(B,median(sample(x,length(x),replace=TRUE)))
quantile(boot_medias,c(alpha/2,1-alpha/2))
     2.5%     97.5% 
 7.683295 11.281443 
quantile(boot_medianas,c(alpha/2,1-alpha/2))
    2.5%    97.5% 
5.816846 9.559478 
hist(boot_medianas,breaks=35,col="#B8D7E8",border="white",
     main="Distribución bootstrap de la mediana",xlab="Mediana")

9. Bootstrap: diferencia de medianas independientes

set.seed(47)
a <- rlnorm(38,2.4,0.45)
b <- rlnorm(41,2.2,0.55)
B <- 4000
boot_d <- replicate(B,{
  aa <- sample(a,length(a),replace=TRUE)
  bb <- sample(b,length(b),replace=TRUE)
  median(aa)-median(bb)
})
c(estimacion=median(a)-median(b),
  quantile(boot_d,c(0.025,0.975)))
estimacion       2.5%      97.5% 
 -1.148896  -4.728635   2.448464

10. Bootstrap pareado: cambio de medianas

set.seed(48)
antes <- rlnorm(30,2.5,0.3)
despues <- antes-rnorm(30,2,0.8)
diferencias <- antes-despues
B <- 4000
boot <- replicate(B,median(sample(diferencias,length(diferencias),replace=TRUE)))
c(estimacion=median(diferencias),quantile(boot,c(0.025,0.975)))
